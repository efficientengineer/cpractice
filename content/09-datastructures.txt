## There is no System.Collections

C has arrays and `malloc`. Everything else you build. That sounds grim, but each structure is small, and building them once teaches you what `List<T>`, `Dictionary<K,V>` and friends actually cost:

| C# | What it is underneath | Built in this chapter |
|---|---|---|
| `List<T>`, `Stack<T>` | a heap array that doubles when full | growable array stack |
| `Queue<T>` | a circular array (ring buffer) | ring buffer queue |
| `LinkedList<T>` | doubly linked nodes | singly and doubly linked lists |
| `Dictionary<K,V>` | hash buckets with chained entries | hash table with string keys |
| `SortedDictionary<K,V>` | a red-black tree (a balanced BST) | plain BST |

The recurring C problems are always the same three: **who owns each node** (and frees it), **how a function changes the caller's pointer**, and **what happens at the empty and one-element edges**.

## Singly linked list

A node holds a value and a pointer to the next node. The list *is* a pointer to the first node; `NULL` is the empty list.

```c
#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int value;
    struct Node *next;     // must say "struct Node": the typedef doesn't exist yet here
} Node;

// The caller's head pointer has to change, so we take its address.
void push_front(Node **head, int value) {
    Node *n = malloc(sizeof *n);
    if (!n) { perror("malloc"); exit(1); }
    n->value = value;
    n->next = *head;
    *head = n;
}

// Returns 1 and stores the value if there was one, 0 if the list was empty.
int pop_front(Node **head, int *out) {
    Node *n = *head;
    if (!n) return 0;
    *out = n->value;
    *head = n->next;
    free(n);
    return 1;
}

void print_list(const Node *n) {
    for (; n; n = n->next) printf("%d ", n->value);
    printf("\n");
}

void free_list(Node *n) {
    while (n) {
        Node *next = n->next;   // read next BEFORE freeing n
        free(n);
        n = next;
    }
}

int main(void) {
    Node *list = NULL;
    for (int i = 1; i <= 4; i++) push_front(&list, i);
    print_list(list);                 // 4 3 2 1
    int v;
    if (pop_front(&list, &v)) printf("popped %d\n", v);
    print_list(list);                 // 3 2 1
    free_list(list);
    return 0;
}
```

Why `Node **head`? C passes everything by value. `push_front(Node *head, ...)` would get a *copy* of the caller's pointer; assigning to it changes the copy and the caller's list never sees the new node (and the node leaks). In C# you'd write `ref Node head`; in C, `ref` is spelled "pass a pointer to it". The alternative style is to return the new head: `list = push_front(list, 5);`.

`sizeof *n` instead of `sizeof(Node)` keeps working if you change `n`'s type. Every `malloc` here should be checked; examples in this chapter sometimes skip it for brevity, real code shouldn't.

### Inserting in sorted order

Walk until the next node is not smaller, then splice in. The empty list and "new smallest" cases need the head to change, which is why a naive version grows special cases:

```c
void insert_sorted(Node **head, int value) {
    Node *n = malloc(sizeof *n);
    n->value = value;
    if (*head == NULL || (*head)->value >= value) {   // special case: new head
        n->next = *head;
        *head = n;
        return;
    }
    Node *cur = *head;
    while (cur->next && cur->next->value < value) cur = cur->next;
    n->next = cur->next;
    cur->next = n;
}
```

### The pointer-to-pointer trick

Notice that "the head pointer" and "some node's `next` field" are both just *a `Node *` somewhere in memory that points at the current node*. If you walk a `Node **` that points at whichever of those links you're on, the head stops being special:

```c
#include <stdio.h>
#include <stdlib.h>

typedef struct Node { int value; struct Node *next; } Node;

void insert_sorted(Node **head, int value) {
    Node **link = head;                        // the link we might rewrite
    while (*link && (*link)->value < value)
        link = &(*link)->next;                 // move to the next node's link
    Node *n = malloc(sizeof *n);
    n->value = value;
    n->next = *link;
    *link = n;                                 // rewrites head OR a next field
}

// Remove the first node holding value. No "previous" pointer, no special case.
int remove_value(Node **head, int value) {
    for (Node **link = head; *link; link = &(*link)->next) {
        if ((*link)->value == value) {
            Node *dead = *link;
            *link = dead->next;
            free(dead);
            return 1;
        }
    }
    return 0;
}

int main(void) {
    Node *list = NULL;
    int xs[] = {5, 1, 4, 2, 3};
    for (int i = 0; i < 5; i++) insert_sorted(&list, xs[i]);
    remove_value(&list, 1);    // the head
    remove_value(&list, 3);    // the middle
    remove_value(&list, 5);    // the tail
    for (Node *n = list; n; n = n->next) printf("%d ", n->value);   // 2 4
    printf("\n");
    while (list) { Node *next = list->next; free(list); list = next; }
    return 0;
}
```

Read `&(*link)->next` as "the address of the `next` field of the node `link` currently points at". Once this pattern clicks, most list code shrinks to a loop and two assignments.

### Reversing in place

Three pointers: what's behind you, where you are, what's ahead. Each step flips one arrow.

```c
Node *reverse(Node *head) {
    Node *prev = NULL;
    while (head) {
        Node *next = head->next;   // save the rest of the list
        head->next = prev;         // flip this node's arrow
        prev = head;
        head = next;
    }
    return prev;                   // the old tail is the new head
}
```

No allocation: the same nodes are relinked. That matters in C because a "reverse" that builds a new list has to free the old one too, or it leaks.

## Doubly linked list, briefly

Add a `prev` pointer and you can remove a node in O(1) given only the node itself, and walk backwards. That is .NET's `LinkedList<T>`. The usual trick is a **sentinel**: a dummy node whose `next` is the first element and whose `prev` is the last. The list is circular through the sentinel, so there are no `NULL` checks at all:

```c
#include <stdio.h>
#include <stdlib.h>

typedef struct DNode { int value; struct DNode *prev, *next; } DNode;

void dl_init(DNode *s) { s->prev = s->next = s; }        // empty: points at itself

void dl_insert_after(DNode *at, int value) {
    DNode *n = malloc(sizeof *n);
    n->value = value;
    n->prev = at;
    n->next = at->next;
    at->next->prev = n;
    at->next = n;
}

void dl_remove(DNode *n) {          // O(1): no searching for the predecessor
    n->prev->next = n->next;
    n->next->prev = n->prev;
    free(n);
}

int main(void) {
    DNode list;                                  // the sentinel lives on the stack
    dl_init(&list);
    for (int i = 1; i <= 3; i++) dl_insert_after(list.prev, i);   // append
    dl_remove(list.next);                        // remove the first element
    for (DNode *n = list.prev; n != &list; n = n->prev) printf("%d ", n->value);
    printf("\n");                                // 3 2 (backwards)
    while (list.next != &list) dl_remove(list.next);
    return 0;
}
```

Linked lists are worse than arrays for almost everything on modern hardware (each node is a separate allocation, and walking them misses the cache). Use them when you need stable node addresses or O(1) splicing, and otherwise reach for a growable array.

> **Performance:** A CPU fetches memory in 64-byte cache lines and prefetches ahead when you walk addresses in order. Iterating a contiguous array of 10,000 entities streams through memory; iterating 10,000 `malloc`'d list nodes is 10,000 potential cache misses (~100 ns each), easily 10-50x slower. In game code, keep things that are updated every frame in arrays. If you need a list, link *array indices* (`int next;`) inside one pre-allocated array instead of `malloc`'ing nodes: that's an object pool with a free list, and it's how most engines store bullets, particles and entities.

## Game entity lists: swap-remove

The bread-and-butter game container is an array of entities that you update every frame and remove dead ones from. Removing from the middle with `memmove` (what `List<T>.RemoveAt` does) is O(n) per removal, so killing many entities in one frame is O(n²). If order doesn't matter, which it usually doesn't for entities, overwrite the dead one with the last element and shrink: O(1).

```c
#include <stdio.h>

typedef struct { int id; float x, vx; int hp; } Entity;

#define MAX_ENTITIES 1024
static Entity entities[MAX_ENTITIES];     // allocated once, up front
static size_t count = 0;

void update(float dt) {
    for (size_t i = 0; i < count; ) {
        Entity *e = &entities[i];
        if (e->hp <= 0) {
            *e = entities[--count];       // swap-remove: move the last one here...
            continue;                      // ...and re-check slot i (it's a new entity)
        }
        e->x += e->vx * dt;
        i++;
    }
}

int main(void) {
    for (int i = 0; i < 6; i++)
        entities[count++] = (Entity){ .id = i, .x = 0, .vx = (float)i, .hp = i % 3 };
    update(0.5f);                          // ids 0 and 3 have hp 0 and are removed
    for (size_t i = 0; i < count; i++) printf("id %d at %.1f\n", entities[i].id, entities[i].x);
    return 0;
}
```

> **Performance:** Notice what is *not* in the frame loop: no `malloc`, no `free`, no building of a new "alive" list. Allocate the entity array once at level load (or statically), with a known maximum, and have `spawn` fail (or recycle the oldest) when it's full. `malloc` per frame costs time, fragments the heap, and makes frame times unpredictable; it's the C version of allocating garbage in Unity's `Update()`. If something needs a stable handle to an entity across swap-removes, give out an index into a separate slot table (or an index plus a generation counter) rather than a raw pointer.

## Stack: the growable array

`List<T>` and `Stack<T>` are an array, a count, and a capacity. When count hits capacity, allocate double and copy. `realloc` does the copying for you:

```c
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int *data;
    size_t len, cap;
} IntStack;

int stack_push(IntStack *s, int v) {
    if (s->len == s->cap) {
        size_t ncap = s->cap ? s->cap * 2 : 8;
        int *p = realloc(s->data, ncap * sizeof *p);
        if (!p) return 0;            // old block is still valid on failure
        s->data = p;
        s->cap = ncap;
    }
    s->data[s->len++] = v;
    return 1;
}

int stack_pop(IntStack *s, int *out) {
    if (s->len == 0) return 0;
    *out = s->data[--s->len];
    return 1;
}

int main(void) {
    IntStack s = {0};                // NULL data, 0 len, 0 cap: a valid empty stack
    for (int i = 0; i < 100; i++) stack_push(&s, i * i);
    int v, sum = 0;
    while (stack_pop(&s, &v)) sum += v;
    printf("sum=%d cap=%zu\n", sum, s.cap);   // sum=328350 cap=128
    free(s.data);
    return 0;
}
```

Doubling makes push *amortized* O(1): n pushes cost about 2n copies in total. Growing by a fixed amount (+10) makes n pushes O(n²). Note `p = realloc(...)` into a temporary: `s->data = realloc(s->data, ...)` loses the old pointer (a leak) if `realloc` fails.

> **Performance:** A growable array that's emptied and refilled every frame (a render list, a list of collisions found this frame) should be *cleared* with `len = 0`, never freed and reallocated. After the first few frames it has reached its working size and never allocates again. Reserve a sensible capacity up front when you know it, exactly like `new List<T>(capacity)`.

## Queue: the ring buffer

A queue needs O(1) at both ends. Shifting an array left on every dequeue is O(n). Instead keep a `head` index and a `len`, and let indices wrap around with `%`:

```c
#include <stdio.h>

#define CAP 4

typedef struct { int data[CAP]; size_t head, len; } Ring;

int ring_push(Ring *r, int v) {
    if (r->len == CAP) return 0;                 // full
    r->data[(r->head + r->len) % CAP] = v;        // tail = head + len, wrapped
    r->len++;
    return 1;
}

int ring_pop(Ring *r, int *out) {
    if (r->len == 0) return 0;
    *out = r->data[r->head];
    r->head = (r->head + 1) % CAP;
    r->len--;
    return 1;
}

int main(void) {
    Ring r = {0};
    int v;
    for (int i = 1; i <= 3; i++) ring_push(&r, i);   // [1 2 3 _]
    ring_pop(&r, &v); ring_pop(&r, &v);              // [_ _ 3 _], head = 2
    for (int i = 4; i <= 6; i++) ring_push(&r, i);   // [5 6 3 4]: wrapped
    printf("full: %d\n", !ring_push(&r, 7));
    while (ring_pop(&r, &v)) printf("%d ", v);       // 3 4 5 6
    printf("\n");
    return 0;
}
```

Games use ring buffers for input event queues, network packet queues, audio sample buffers, and "last N frames" histories (frame-time graphs, replay buffers). They never allocate after setup, which is exactly what you want in a frame loop.

Growing a ring buffer is the subtle part: after a wrap, the elements are in two pieces (`head..cap-1` then `0..tail-1`), so a plain `realloc` puts a gap in the middle. You must copy them out in order. You'll do that in an exercise. (`Queue<T>` in .NET does exactly this.) With a power-of-two capacity, `% cap` can become `& (cap - 1)`, which is why many ring buffers insist on powers of two.

## Hash table with chaining

`Dictionary<string,int>` is: an array of buckets; hash the key to pick a bucket; each bucket is a short linked list of entries whose keys hashed there. Lookup is "hash, then compare keys along one short chain". Keep chains short by growing the bucket array when `count / nbuckets` (the *load factor*) gets too high.

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct Entry {
    char *key;                // owned copy
    int value;
    struct Entry *next;       // next entry in the same bucket
} Entry;

typedef struct {
    Entry **buckets;          // array of chain heads
    size_t nbuckets, count;
} Map;

// FNV-1a: simple, decent spread. Unsigned so overflow wraps (well defined).
static size_t hash_str(const char *s) {
    size_t h = 1469598103934665603u;
    for (; *s; s++) {
        h ^= (unsigned char)*s;
        h *= 1099511628211u;
    }
    return h;
}

static char *copy_str(const char *s) {   // strdup is POSIX/C23, not C17
    size_t n = strlen(s) + 1;
    char *p = malloc(n);
    if (p) memcpy(p, s, n);
    return p;
}

Map *map_new(size_t nbuckets) {
    Map *m = malloc(sizeof *m);
    m->buckets = calloc(nbuckets, sizeof *m->buckets);   // all chains empty (NULL)
    m->nbuckets = nbuckets;
    m->count = 0;
    return m;
}

int *map_get(Map *m, const char *key) {
    for (Entry *e = m->buckets[hash_str(key) % m->nbuckets]; e; e = e->next)
        if (strcmp(e->key, key) == 0) return &e->value;
    return NULL;
}

void map_put(Map *m, const char *key, int value) {
    int *slot = map_get(m, key);
    if (slot) { *slot = value; return; }                 // overwrite, like dict[key] = v
    size_t b = hash_str(key) % m->nbuckets;
    Entry *e = malloc(sizeof *e);
    e->key = copy_str(key);
    e->value = value;
    e->next = m->buckets[b];                             // push onto the chain
    m->buckets[b] = e;
    m->count++;
}

void map_free(Map *m) {
    for (size_t i = 0; i < m->nbuckets; i++) {
        Entry *e = m->buckets[i];
        while (e) {
            Entry *next = e->next;
            free(e->key);                                // free what the entry owns...
            free(e);                                     // ...then the entry
            e = next;
        }
    }
    free(m->buckets);
    free(m);
}

int main(void) {
    Map *m = map_new(16);
    const char *words[] = {"the", "cat", "sat", "on", "the", "mat", "the", "end"};
    for (int i = 0; i < 8; i++) {
        int *c = map_get(m, words[i]);
        map_put(m, words[i], c ? *c + 1 : 1);
    }
    printf("the=%d cat=%d dog=%s count=%zu\n",
           *map_get(m, "the"), *map_get(m, "cat"),
           map_get(m, "dog") ? "yes" : "missing", m->count);
    map_free(m);
    return 0;
}
```

Things C makes you decide that C# hides:

- **Key ownership.** The map copies the key. If it stored the caller's pointer, a caller who reused a buffer (`char buf[64]` in a read loop) would silently change every key. Copy on insert, free on removal.
- **Returning "not found".** `TryGetValue` returns `bool` with an `out` parameter; here `map_get` returns a pointer to the value or `NULL`. Returning a pointer into the table is handy but it dangles after the entry is removed or the table is resized.
- **Resizing.** To grow, allocate a bigger bucket array and move every entry to `hash % new_size`. Entries are relinked, not copied, so it's a loop over chains. Keep the load factor around 1 or below and lookups stay O(1) on average.
- **Hash quality.** `hash % nbuckets` with a weak hash and a power-of-two bucket count uses only the low bits. FNV-1a mixes reasonably; summing characters does not ("ab" and "ba" collide).

> **Performance:** Hashing a string and comparing it is far more expensive than comparing two ints. Don't look up `"health_potion"` by name every frame: resolve names to small integer ids (or direct pointers) once at load time and use those in the hot path. And a linear search through an array of names is fine for 10 items but turns into O(n²) when done once per item per frame. Chained tables like this one cost a pointer chase per entry; production tables often use *open addressing* (entries stored directly in one array, probing to the next slot on collision), which is friendlier to the cache.

## Binary search tree

Each node's left subtree holds smaller keys, the right subtree larger. Insert and find walk one path from the root, so they're O(height): O(log n) if the tree stays bushy, O(n) if you insert sorted data and it degenerates into a list. Production trees (red-black, AVL; `SortedDictionary`) rebalance on insert. A plain BST is the core of them all.

```c
#include <stdio.h>
#include <stdlib.h>

typedef struct Tree {
    int key;
    struct Tree *left, *right;
} Tree;

// Recursive insert that returns the (possibly new) subtree root.
Tree *insert(Tree *t, int key) {
    if (!t) {
        t = malloc(sizeof *t);
        t->key = key;
        t->left = t->right = NULL;
    } else if (key < t->key) {
        t->left = insert(t->left, key);
    } else if (key > t->key) {
        t->right = insert(t->right, key);
    }                                     // equal: already present, do nothing
    return t;
}

// Iterative find: no recursion needed for one path.
const Tree *find(const Tree *t, int key) {
    while (t && t->key != key) t = key < t->key ? t->left : t->right;
    return t;
}

// In-order: left, self, right. Visits keys in sorted order.
void print_inorder(const Tree *t) {
    if (!t) return;
    print_inorder(t->left);
    printf("%d ", t->key);
    print_inorder(t->right);
}

// Post-order free: children first, because after free(t) you may not read t->left.
void free_tree(Tree *t) {
    if (!t) return;
    free_tree(t->left);
    free_tree(t->right);
    free(t);
}

int main(void) {
    Tree *root = NULL;
    int keys[] = {50, 30, 70, 20, 40, 60, 80, 40};
    for (int i = 0; i < 8; i++) root = insert(root, keys[i]);
    print_inorder(root);                                   // 20 30 40 50 60 70 80
    printf("\nfind 60: %s, find 65: %s\n",
           find(root, 60) ? "yes" : "no", find(root, 65) ? "yes" : "no");
    free_tree(root);
    return 0;
}
```

The insert can also be written with the pointer-to-pointer trick: walk `Tree **link = &root` down to the `NULL` child slot, then `*link = new_node`. Same idea as the list.

> **Performance:** Trees are pointer-chasing structures too. For data that's built once and queried often (a sorted list of item ids, a leaderboard snapshot), a sorted array with binary search does the same O(log n) lookups with no per-node allocations and far better cache behavior. For "what's near this point?" questions in a game, neither is right: use a uniform grid or spatial hash, which turns an O(n²) all-pairs check into roughly O(n).

## Freeing recursive structures

Every structure above owns heap memory in layers. The rule: **free what something owns before you free it, and never touch a node after freeing it.**

- Lists: save `next` first, then free. `for (n = head; n; n = n->next) free(n);` reads `n->next` from freed memory (AddressSanitizer reports *heap-use-after-free*).
- Entries that own strings: free the key, then the entry.
- Trees: post-order (children, then the node). Recursion depth equals tree height, so a degenerate tree with a million nodes can overflow the stack; iterative frees (or balanced trees) avoid that.
- Give every structure one `x_free` function and make it accept `NULL`, like `free` itself. Then error paths can call it unconditionally.

C# never makes you think about this; the GC traces from roots and frees unreachable nodes in bulk. In C, if you lose the last pointer to a node without freeing it, it's leaked for the life of the process. This site's sanitizer reports leaks, so you'll find out.

#include <stdlib.h>

typedef struct Node {
    int value;
    struct Node *next;
} Node;

static Node *new_node(int value, Node *next) {
    Node *n = malloc(sizeof *n);
    if (!n) abort();
    n->value = value;
    n->next = next;
    return n;
}

void push_front(Node **head, int value) {
    *head = new_node(value, *head);
}

void push_back(Node **head, int value) {
    Node **link = head;
    while (*link) link = &(*link)->next;
    *link = new_node(value, NULL);
}

int pop_front(Node **head, int *out) {
    Node *n = *head;
    if (!n) return 0;
    *out = n->value;
    *head = n->next;
    free(n);
    return 1;
}

size_t list_len(const Node *head) {
    size_t n = 0;
    for (; head; head = head->next) n++;
    return n;
}

void list_free(Node **head) {
    Node *n = *head;
    while (n) {
        Node *next = n->next;
        free(n);
        n = next;
    }
    *head = NULL;
}

static int t_is(const Node *h, const int *want, size_t n) {
    for (size_t i = 0; i < n; i++, h = h->next)
        if (!h || h->value != want[i]) return 0;
    return h == NULL;
}

int main(void) {
    Node *h = NULL;
    int v = -99;
    CHECK_INT(list_len(h), 0);
    CHECK_INT(pop_front(&h, &v), 0);
    CHECK_INT(v, -99);

    push_back(&h, 5);
    CHECK(t_is(h, (int[]){5}, 1));
    push_front(&h, 3);
    push_front(&h, 1);
    push_back(&h, 7);
    push_back(&h, 9);

title: Sorted insert and remove-all
--- prompt

--- solution
#include <stdlib.h>

typedef struct Node {
    int value;
    struct Node *next;
} Node;

void insert_sorted(Node **head, int value) {
    Node **link = head;
    while (*link && (*link)->value < value) link = &(*link)->next;
    Node *n = malloc(sizeof *n);
    if (!n) abort();
    n->value = value;
    n->next = *link;
    *link = n;
}

size_t remove_all(Node **head, int value) {
    size_t removed = 0;
    Node **link = head;
    while (*link) {
        if ((*link)->value == value) {
            Node *dead = *link;
            *link = dead->next;     // don't advance: *link is now the next node
            free(dead);
            removed++;
        } else {
            link = &(*link)->next;
        }
    }
    return removed;
}
--- hint
For insertion, stop at the first link whose node is not smaller than `value` (or is `NULL`), then splice the new node in there.
--- hint
In `remove_all`, after unlinking a node do **not** advance `link`: it already points at the link that now holds the following node. Only advance when you keep the node.
--- harness

    return h == NULL;
}

    while (h) { Node *n = h->next; free(h); h = n; }

=== exercise reversemerge
title: Reverse and merge without allocating
--- prompt
Write two functions that only relink existing nodes. They must not call `malloc` or `free`: the tests free the result list themselves, so any node you create or drop is reported as a leak.

- `Node *list_reverse(Node *head)`: reverse the list in place and return the new head.
- `Node *list_merge(Node *a, Node *b)`: `a` and `b` are each sorted ascending. Merge them into one ascending list and return its head. When values are equal, nodes from `a` come first (a stable merge, like the merge step of merge sort).

For the merge, a `Node **tail` that always points at the link to fill next keeps the code short.
--- starter
#include <stdlib.h>

typedef struct Node {
    int value;
    struct Node *next;
} Node;

    return head;
}

Node *list_merge(Node *a, Node *b) {
    (void)b;
    return a;
}
--- solution
#include <stdlib.h>

typedef struct Node {
    int value;
    struct Node *next;
} Node;

Node *list_reverse(Node *head) {
    Node *prev = NULL;
    while (head) {
        Node *next = head->next;
        head->next = prev;
        prev = head;
        head = next;
    }
    return prev;
}

Node *list_merge(Node *a, Node *b) {
    Node *result = NULL;
    Node **tail = &result;
    while (a && b) {
        if (b->value < a->value) {
            *tail = b;
            b = b->next;
        } else {
            *tail = a;
            a = a->next;
        }
        tail = &(*tail)->next;
    }
    *tail = a ? a : b;
    return result;
}
--- hint
Reverse: keep `prev` (initially `NULL`). For each node, save `next`, point the node at `prev`, then advance both.
--- hint
Merge: start with `Node *result = NULL; Node **tail = &result;`. Each step, attach the smaller front node to `*tail`, advance that list, then `tail = &(*tail)->next`. When one list runs out, attach the rest of the other in one assignment.
--- hint
Stability: take from `b` only when `b->value < a->value`, strictly.
--- harness

=== exercise entities
title: Entity list with swap-remove
--- prompt
Write the per-frame entity container from the chapter. The world allocates its storage **once** and never reallocates, so pointers into it stay valid between frames as long as nothing is removed.

```c
typedef struct { int id; float x, y, vx, vy; int hp; } Entity;
typedef struct { Entity *items; size_t len, cap; } World;
```

- `int world_init(World *w, size_t cap)`: allocate room for `cap` entities in one allocation; return 0 on success, -1 on failure.
- `Entity *world_spawn(World *w, Entity e)`: copy `e` in and return a pointer to the stored copy, or `NULL` if the world is full. Never grow.
- `void world_update(World *w, float dt)`: remove every entity with `hp <= 0`, and move every survivor by `vx * dt`, `vy * dt`. Order of survivors doesn't matter: use **swap-remove**.
- `void world_free(World *w)`.

The tests run a frame where 150,000 of 300,000 entities die. Removing each with `memmove` (shifting everything after it) is O(n²) and times out.
--- starter
#include <stdlib.h>
#include <string.h>

typedef struct { int id; float x, y, vx, vy; int hp; } Entity;
typedef struct { Entity *items; size_t len, cap; } World;

int world_init(World *w, size_t cap) {
    w->items = NULL;
    w->len = 0;
    w->cap = cap;
    return -1;
}

    (void)w; (void)e;
    return NULL;
}

    for (size_t i = 0; i < w->len; ) {
        Entity *e = &w->items[i];
        if (e->hp <= 0) {
            memmove(e, e + 1, (w->len - i - 1) * sizeof *e);   /* O(n) per removal! */
            w->len--;
            continue;
        }
        e->x += e->vx * dt;
        e->y += e->vy * dt;
        i++;
    }
}

void world_free(World *w) {
    free(w->items);
}
--- solution
#include <stdlib.h>
#include <string.h>

typedef struct { int id; float x, y, vx, vy; int hp; } Entity;
typedef struct { Entity *items; size_t len, cap; } World;

int world_init(World *w, size_t cap) {
    w->items = malloc(cap * sizeof *w->items);
    w->len = 0;
    w->cap = w->items ? cap : 0;
    return w->items ? 0 : -1;
}

Entity *world_spawn(World *w, Entity e) {
    if (w->len == w->cap) return NULL;
    w->items[w->len] = e;
    return &w->items[w->len++];
}

void world_update(World *w, float dt) {
    for (size_t i = 0; i < w->len; ) {
        Entity *e = &w->items[i];
        if (e->hp <= 0) {
            *e = w->items[--w->len];
            continue;
        }
        e->x += e->vx * dt;
        e->y += e->vy * dt;
        i++;
    }
}

void world_free(World *w) {
    free(w->items);
    w->items = NULL;
    w->len = w->cap = 0;
}
--- hint
Swap-remove: `w->items[i] = w->items[--w->len];` and then *don't* advance `i`, because slot `i` now holds a different entity that still needs checking (and moving).
--- hint

--- harness

--- solution
#include <stdlib.h>
#include <string.h>

typedef struct {
    int *data;
    size_t cap;
    size_t head;
    size_t len;
} Queue;

Queue *q_new(size_t cap) {
    if (cap == 0) cap = 1;
    Queue *q = malloc(sizeof *q);
    if (!q) return NULL;
    q->data = malloc(cap * sizeof *q->data);
    if (!q->data) { free(q); return NULL; }
    q->cap = cap;
    q->head = 0;
    q->len = 0;
    return q;
}

static void grow(Queue *q) {
    size_t ncap = q->cap * 2;
    int *nd = malloc(ncap * sizeof *nd);
    if (!nd) abort();
    for (size_t i = 0; i < q->len; i++)
        nd[i] = q->data[(q->head + i) % q->cap];
    free(q->data);
    q->data = nd;
    q->cap = ncap;
    q->head = 0;
}

void q_push(Queue *q, int v) {
    if (q->len == q->cap) grow(q);
    q->data[(q->head + q->len) % q->cap] = v;
    q->len++;
}

int q_pop(Queue *q, int *out) {
    if (q->len == 0) return 0;
    *out = q->data[q->head];
    q->head = (q->head + 1) % q->cap;
    q->len--;
    return 1;
}

int q_peek(const Queue *q, int *out) {
    if (q->len == 0) return 0;
    *out = q->data[q->head];
    return 1;
}

size_t q_len(const Queue *q) {
    return q->len;
}

void q_free(Queue *q) {
    if (!q) return;
    free(q->data);
    free(q);
}
--- hint
The back of the queue is at `(head + len) % cap`. The front is at `head`, and popping advances `head` with wrap-around.
--- hint
Easiest grow: allocate a fresh array of `2 * cap`, copy element `i` from `data[(head + i) % cap]` to `new[i]` for `i < len`, free the old array, set `head = 0`.
--- harness

    CHECK_INT(q_pop(q, &v), 1); CHECK_INT(v, 1);
    CHECK_INT(q_pop(q, &v), 1); CHECK_INT(v, 2);
    q_push(q, 4); q_push(q, 5); q_push(q, 6);   /* wraps around */
    CHECK_INT(q_len(q), 4);
    CHECK_INT(q_peek(q, &v), 1); CHECK_INT(v, 3);
    q_push(q, 7);                               /* full and wrapped: must grow correctly */
    q_push(q, 8);
    CHECK_INT(q_len(q), 6);
    int want[] = {3, 4, 5, 6, 7, 8}, ok = 1;
    for (int i = 0; i < 6; i++) if (!q_pop(q, &v) || v != want[i]) ok = 0;
    CHECK(ok);
    CHECK_INT(q_len(q), 0);

    q = q_new(1);
    int next_in = 0, next_out = 0;
    ok = 1;
    for (int round = 0; round < 50; round++) {      /* interleave, wrapping at every size */
        for (int k = 0; k < round + 3; k++) q_push(q, next_in++);

#include <stdlib.h>
#include <string.h>

typedef struct Entry {
    char *key;
    int value;
    struct Entry *next;
} Entry;

typedef struct {
    Entry **buckets;
    size_t nbuckets;
    size_t count;
} Map;

static size_t hash_str(const char *s) {
    size_t h = 1469598103934665603u;
    for (; *s; s++) {
        h ^= (unsigned char)*s;
        h *= 1099511628211u;
    }
    return h;
}

static char *copy_str(const char *s) {
    size_t n = strlen(s) + 1;
    char *p = malloc(n);
    if (!p) abort();
    memcpy(p, s, n);
    return p;
}

Map *map_new(void) {
    Map *m = malloc(sizeof *m);
    if (!m) abort();
    m->nbuckets = 8;
    m->count = 0;
    m->buckets = calloc(m->nbuckets, sizeof *m->buckets);
    if (!m->buckets) abort();
    return m;
}

static Entry **find_link(const Map *m, const char *key) {
    Entry **link = &m->buckets[hash_str(key) % m->nbuckets];

}

static void grow(Map *m) {
    size_t nb = m->nbuckets * 2;

}

}

#include <stdlib.h>
#include <string.h>

typedef struct Entry {
    char *key;
    int value;
    struct Entry *next;
} Entry;

typedef struct {
    Entry **buckets;
    size_t nbuckets;
    size_t count;
} Map;

static size_t hash_str(const char *s) {
    size_t h = 1469598103934665603u;
    for (; *s; s++) {
        h ^= (unsigned char)*s;
        h *= 1099511628211u;
    }
    return h;
}

static char *copy_str(const char *s) {
    size_t n = strlen(s) + 1;
    char *p = malloc(n);
    if (!p) abort();
    memcpy(p, s, n);
    return p;
}

Map *map_new(void) {
    Map *m = malloc(sizeof *m);

--- solution
#include <stdlib.h>

typedef struct Tree {
    int key;
    struct Tree *left, *right;
} Tree;

Tree *bst_insert(Tree *root, int key) {
    Tree **link = &root;
    while (*link) {
        if (key == (*link)->key) return root;
        link = key < (*link)->key ? &(*link)->left : &(*link)->right;
    }
    Tree *n = malloc(sizeof *n);
    if (!n) abort();
    n->key = key;
    n->left = n->right = NULL;
    *link = n;
    return root;
}

int bst_contains(const Tree *root, int key) {
    while (root && root->key != key) root = key < root->key ? root->left : root->right;
    return root != NULL;
}

int bst_height(const Tree *root) {
    if (!root) return 0;
    int l = bst_height(root->left), r = bst_height(root->right);
    return 1 + (l > r ? l : r);
}

size_t bst_to_array(const Tree *root, int *out) {
    if (!root) return 0;
    size_t n = bst_to_array(root->left, out);
    out[n++] = root->key;
    return n + bst_to_array(root->right, out + n);
}

Tree *bst_remove(Tree *root, int key) {
    if (!root) return NULL;
    if (key < root->key) {
        root->left = bst_remove(root->left, key);
    } else if (key > root->key) {
        root->right = bst_remove(root->right, key);
    } else if (!root->left || !root->right) {
        Tree *child = root->left ? root->left : root->right;
        free(root);
        return child;
    } else {
        Tree *m = root->right;
        while (m->left) m = m->left;
        root->key = m->key;
        root->right = bst_remove(root->right, m->key);
    }
    return root;
}

void bst_free(Tree *root) {
    if (!root) return;
    bst_free(root->left);
    bst_free(root->right);
    free(root);
}
--- hint
`bst_to_array` can return a count: fill the left subtree into `out`, getting `n`; put the root at `out[n]`; then fill the right subtree starting at `out + n + 1`.
--- hint
`bst_remove` is easiest recursively, returning the new subtree root: `root->left = bst_remove(root->left, key)`. A node with zero or one child is replaced by that child (or `NULL`) after freeing it.
--- hint
Two children: find the leftmost node of the right subtree (its minimum), copy its key into this node, then `root->right = bst_remove(root->right, that_key)`. That removal hits the easy case because the minimum has no left child.
--- harness

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

typedef struct WNode {
    char *word;
    int count;
    struct WNode *left, *right;
} WNode;

static WNode *add(WNode *t, const char *w) {
    if (!t) {
        t = malloc(sizeof *t);
        size_t n = strlen(w) + 1;
        t->word = malloc(n);
        memcpy(t->word, w, n);
        t->count = 1;
        t->left = t->right = NULL;
        return t;
    }
    int c = strcmp(w, t->word);
    if (c < 0) t->left = add(t->left, w);
    else if (c > 0) t->right = add(t->right, w);
    else t->count++;
    return t;
}

static void print_tree(const WNode *t) {
    if (!t) return;
    print_tree(t->left);
    printf("%s %d\n", t->word, t->count);
    print_tree(t->right);
}

static void free_tree(WNode *t) {
    if (!t) return;
    free_tree(t->left);
    free_tree(t->right);
    free(t->word);
    free(t);
}

int main(void) {
    char buf[64];
    WNode *root = NULL;
    while (scanf("%63s", buf) == 1) {
        for (char *p = buf; *p; p++) *p = (char)tolower((unsigned char)*p);
        root = add(root, buf);
    }
    print_tree(root);
    free_tree(root);
    return 0;
}

--- q
Keys are inserted into an empty BST in the order 5, 3, 8, 1, 4, 9. What does a **pre-order** traversal (node, then left, then right) print?
--- code
#include <stdio.h>
#include <stdlib.h>

typedef struct T { int k; struct T *l, *r; } T;

T *ins(T *t, int k) {
    if (!t) { t = calloc(1, sizeof *t); t->k = k; }
    else if (k < t->k) t->l = ins(t->l, k);
    else t->r = ins(t->r, k);
    return t;
}

void pre(T *t) {
    if (!t) return;
    printf("%d ", t->k);
    pre(t->l);
    pre(t->r);
}

void del(T *t) { if (t) { del(t->l); del(t->r); free(t); } }

int main(void) {
    T *root = NULL;
    int ks[] = {5, 3, 8, 1, 4, 9};
    for (int i = 0; i < 6; i++) root = ins(root, ks[i]);
    pre(root);
    printf("\n");
    del(root);
    return 0;
}
--- choice
1 3 4 5 8 9
--- answer
5 3 1 4 8 9
--- choice
1 4 3 9 8 5
--- choice
5 3 8 1 4 9
--- explain
Pre-order visits a node before its subtrees: 5, then the whole left subtree (3, 1, 4), then the right (8, 9). In-order would give the sorted order; post-order (1 4 3 9 8 5) is the order you free in.
