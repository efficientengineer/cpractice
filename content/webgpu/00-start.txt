=== exercise double
title: Double
--- prompt
Write `async function double(device, data)` that doubles a Uint32Array on the GPU.
--- starter
async function double(device, data) { return data; }
--- solution
async function double(device, data) {
  const module = device.createShaderModule({ code: `
    @group(0) @binding(0) var<storage, read_write> d: array<u32>;
    @compute @workgroup_size(64) fn main(@builtin(global_invocation_id) id: vec3u) {
      if (id.x < arrayLength(&d)) { d[id.x] *= 2u; }
    }` });
  const pipeline = device.createComputePipeline({ layout: 'auto', compute: { module } });
  const buf = device.createBuffer({ size: data.byteLength, usage: GPUBufferUsage.STORAGE | GPUBufferUsage.COPY_SRC | GPUBufferUsage.COPY_DST });
  device.queue.writeBuffer(buf, 0, data);
  const bg = device.createBindGroup({ layout: pipeline.getBindGroupLayout(0), entries: [{ binding: 0, resource: { buffer: buf } }] });
  const enc = device.createCommandEncoder(); const pass = enc.beginComputePass();
  pass.setPipeline(pipeline); pass.setBindGroup(0, bg); pass.dispatchWorkgroups(Math.ceil(data.length / 64)); pass.end();
  device.queue.submit([enc.finish()]);
  return new Uint32Array(await readBuffer(device, buf));
}
--- harness
const device = await getDevice();
eq([...await double(device, new Uint32Array([1, 2, 3, 100]))], [2, 4, 6, 200]);
