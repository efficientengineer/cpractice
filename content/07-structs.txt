title: Structs, enums, unions
blurb: C's only way to build your own types: value-semantics structs, padding and layout, enums as plain ints, tagged unions, flexible array members, and methods without classes.

## A struct is a C# struct without the methods

A `struct` groups named fields into one type. The closest relative in your toolbox is a C# `struct`: a value type, copied on assignment, laid out inline. There are no methods, constructors, properties, access modifiers, inheritance or interfaces. Just data.

```c
#include <stdio.h>

struct point {
    int x;
    int y;
};

int main(void) {
    struct point p;          // the type's name is "struct point", both words
    p.x = 3;
    p.y = 4;
    struct point q = {1, 2}; // positional initializer, in field order
    printf("(%d, %d) (%d, %d)\n", p.x, p.y, q.x, q.y);
    return 0;
}
```

The tag `point` lives in its own namespace, which is why you must write `struct point` every time. Most code adds a `typedef` to give the type a one-word name:

```c
typedef struct {
    double x, y;
} Vec2;                         // anonymous struct, named via typedef

typedef struct node {           // tagged too: needed when it refers to itself
    int value;
    struct node *next;          // "Node" does not exist yet on this line
} Node;
```

Both styles are common. The Linux kernel avoids typedefs for structs on purpose (seeing `struct` tells you it is a struct); most application code uses them. Pick one per project.

## Initialization

```c
#include <stdio.h>

typedef struct {
    const char *name;
    int hp;
    int x, y;
    double speed;
} Player;

int main(void) {
    Player a = {"Ada", 100, 0, 0, 1.5};             // positional: fragile if fields move
    Player b = {.name = "Bob", .hp = 80, .speed = 2.0};   // designated: x and y become 0
    Player c = {0};                                  // everything zero / NULL
    Player d;                                        // local: garbage, like any local

    printf("%s %d %d %.1f\n", b.name, b.hp, b.x, b.speed);
    printf("%d %p\n", c.hp, (void *)c.name);

    // A compound literal: an unnamed struct value, created on the spot
    d = (Player){.name = "Dee", .hp = 50};
    printf("%s %d\n", d.name, d.hp);
    (void)a;
    return 0;
}
```

**Designated initializers** (`.field = value`) are the C equivalent of C#'s object initializers, and as with arrays, any field you leave out is zeroed. Prefer them: the code stays correct when someone reorders or adds fields.

**Compound literals** `(Type){ ... }` create a temporary value of any struct (or array) type. They are how you "construct" a value inline: assigning, passing to a function, or returning.

## Value semantics

Assignment copies every byte of a struct. Passing one to a function copies it. Returning one copies it. This includes arrays *inside* a struct, which is the one way to copy or return a whole array with `=`.

```c
#include <stdio.h>

typedef struct {
    int scores[3];
    int count;
} Stats;

void bump(Stats s) {           // s is a copy (gcc even warns the write is pointless)
    s.scores[0] = 999;
}

Stats make(void) {
    return (Stats){.scores = {7, 8, 9}, .count = 3};   // returning a struct: fine
}

int main(void) {
    Stats a = make();
    Stats b = a;               // copies the whole array too
    b.scores[0] = 1;
    bump(a);
    printf("%d %d\n", a.scores[0], b.scores[0]);   // 7 1
    return 0;
}
```

Structs cannot be compared with `==`. Compare field by field. (`memcmp` is tempting but wrong in general: padding bytes, below, may hold garbage.)

## Pointers to structs and ->

Copying a large struct on every call is wasteful, and a copy cannot be modified on the caller's behalf. So, as with C# `ref` parameters, you pass a pointer. `p->field` is shorthand for `(*p).field`:

```c
#include <stdio.h>

typedef struct {
    double x, y;
} Vec2;

void scale(Vec2 *v, double k) {    // modifies the caller's value
    v->x *= k;
    v->y *= k;
}

double dot(const Vec2 *a, const Vec2 *b) {   // const: read-only, no copy
    return a->x * b->x + a->y * b->y;
}

int main(void) {
    Vec2 v = {3, 4};
    scale(&v, 2);
    Vec2 *p = &v;
    printf("%.1f %.1f %.1f\n", p->x, (*p).y, dot(&v, &v));
    return 0;
}
```

The parentheses in `(*p).x` are required because `.` binds tighter than `*`: `*p.x` means `*(p.x)`. That ugliness is why `->` exists.

Rule of thumb: pass small structs (a couple of numbers) by value, everything else by `const T *` for reading or `T *` for modifying.

## Padding and alignment

Each type has an *alignment*: an `int` wants an address divisible by 4, a `double` or pointer by 8. To satisfy that, the compiler inserts invisible padding between fields, and at the end so that arrays of the struct stay aligned. Fields are never reordered.

```c
#include <stdio.h>
#include <stddef.h>   // offsetof

struct bad {
    char a;      // offset 0, then 7 bytes of padding
    double b;    // offset 8
    char c;      // offset 16, then 3 bytes of padding
    int d;       // offset 20
};               // size 24

struct good {
    double b;    // offset 0
    int d;       // offset 8
    char a;      // offset 12
    char c;      // offset 13, then 2 bytes of tail padding
};               // size 16

int main(void) {
    printf("bad:  %zu bytes, b at %zu, c at %zu, d at %zu\n",
           sizeof(struct bad), offsetof(struct bad, b), offsetof(struct bad, c), offsetof(struct bad, d));
    printf("good: %zu bytes\n", sizeof(struct good));
    return 0;
}
```

The rule that gets you the minimum size almost every time: **order fields from largest alignment to smallest**. It matters when you have a million of them in an array, or when the struct crosses a boundary (a file format, a network packet, a GPU buffer). For those cases never assume a layout: write fields out one by one, and check with `sizeof` and `offsetof` (or `_Static_assert(sizeof(struct good) == 16, "layout")`).

> **Performance:** padding is wasted cache. A vertex declared as `{ char flags; float pos[3]; char material; float uv[2]; }` is 28 bytes, of which 6 carry nothing; reordered largest-first it is 24. Across a million vertices that is 4 MB of memory traffic per pass for no information. Before shipping a struct you store by the thousand, print its `sizeof`, and pin the size with `_Static_assert` so a later edit cannot quietly grow it. Also choose field types for storage: a normal often fits in three `int8_t`s or a packed 32-bit value, a color in four `uint8_t`s, a UV in two `uint16_t`s.

## Nesting and arrays of structs

Structs nest by value (the inner struct is stored inline, not as a reference), and arrays of structs are contiguous:

```c
#include <stdio.h>

typedef struct { int x, y; } Point;
typedef struct { Point min, max; } Box;   // 16 bytes, two Points inline

typedef struct {
    char name[16];      // a fixed buffer, stored inside the struct
    int price;
} Item;

int main(void) {
    Box b = {.min = {0, 0}, .max = {10, 5}};
    b.max.x += 2;
    printf("%d x %d\n", b.max.x - b.min.x, b.max.y - b.min.y);

    Item shop[] = {
        {"sword", 120},
        {"shield", 90},
        {.name = "potion", .price = 15},
    };
    int n = sizeof shop / sizeof shop[0];
    int total = 0;
    for (Item *it = shop; it < shop + n; it++)    // walk with a pointer
        total += it->price;
    printf("%d items, total %d, first %s\n", n, total, shop[0].name);
    return 0;
}
```

`char name[16]` puts the characters inside the struct, so copying the struct copies the name. A `const char *name` field would only copy the pointer, with both copies pointing at the same characters. That distinction (deep vs shallow copy) is the same one you know from C#, just more visible.

### Array of structs, or struct of arrays?

An array of structs (AoS) keeps each object's fields together. A struct of arrays (SoA) keeps each *field* together, one array per field:

```c
#include <stdio.h>

#define MAXP 4

typedef struct { float x, y, vx, vy, life; unsigned color; } Particle;   // AoS element: 24 bytes
Particle aos[MAXP];

typedef struct {                                                       // SoA: one array per field
    float x[MAXP], y[MAXP], vx[MAXP], vy[MAXP], life[MAXP];
    unsigned color[MAXP];
} Particles;
Particles soa;

int main(void) {
    for (int i = 0; i < MAXP; i++) {
        aos[i] = (Particle){.vx = 1, .vy = 2};
        soa.vx[i] = 1; soa.vy[i] = 2;
    }
    float dt = 0.5f;
    for (int i = 0; i < MAXP; i++) {      // AoS: reads all 24 bytes of each particle
        aos[i].x += aos[i].vx * dt;
        aos[i].y += aos[i].vy * dt;
    }
    for (int i = 0; i < MAXP; i++) {      // SoA: four tight streams, nothing else touched
        soa.x[i] += soa.vx[i] * dt;
        soa.y[i] += soa.vy[i] * dt;
    }
    printf("%.1f %.1f | %.1f %.1f\n", aos[3].x, aos[3].y, soa.x[3], soa.y[3]);
    return 0;
}
```

> **Performance:** the movement loop needs 16 of the 24 bytes per particle. With AoS the other 8 (life, color) still come through the cache; with SoA every byte loaded is used, and the loop is exactly the shape the compiler vectorizes (8 floats per instruction). Use AoS for things you handle one at a time or always use whole (a GPU vertex, a config record), and SoA for big collections that hot loops sweep a few fields of (particles, positions for culling, pixels split into channels). Splitting rarely used ("cold") fields out of a hot struct is the halfway house.

## Enums are ints

```c
enum color { RED, GREEN, BLUE };          // 0, 1, 2
enum http { OK = 200, NOT_FOUND = 404, TEAPOT = 418 };
typedef enum { DIR_N, DIR_E, DIR_S, DIR_W, DIR_COUNT } Dir;
```

Unlike C# enums, the constants are plain `int`s in the enclosing scope (no `Color.Red`, so prefix them by hand: `DIR_N`), and an `enum` variable accepts any integer without complaint. There is no `ToString()`; the usual trick is a parallel array of names indexed by the value. A trailing `..._COUNT` constant gives you the number of values for free.

```c
#include <stdio.h>

typedef enum { DIR_N, DIR_E, DIR_S, DIR_W, DIR_COUNT } Dir;
static const char *dir_names[DIR_COUNT] = {"north", "east", "south", "west"};

Dir turn_right(Dir d) { return (Dir)((d + 1) % DIR_COUNT); }

int main(void) {
    Dir d = DIR_W;
    printf("%s -> %s\n", dir_names[d], dir_names[turn_right(d)]);
    switch (d) {               // gcc -Wall warns if a switch on an enum misses a value
    case DIR_N: case DIR_S: puts("vertical"); break;
    case DIR_E: case DIR_W: puts("horizontal"); break;
    default: break;
    }
    return 0;
}
```

## Unions and tagged unions

A `union` looks like a struct, but all its fields share the same memory. Its size is that of its largest member, and only the member you last wrote holds a meaningful value. On its own that is a low-level tool. Paired with an enum saying which member is live, it becomes a **tagged union**: C's version of a discriminated union, F#'s `type Shape = Circle of float | Rect of ...`, or a C# class hierarchy with a `switch` on the type.

```c
#include <stdio.h>

typedef enum { VAL_INT, VAL_DOUBLE, VAL_STR } ValKind;

typedef struct {
    ValKind kind;          // the tag: which union member is valid
    union {                // anonymous union (C11): members accessed as v.i, v.d, v.s
        int i;
        double d;
        const char *s;
    };
} Value;

void print_value(const Value *v) {
    switch (v->kind) {
    case VAL_INT:    printf("int %d\n", v->i); break;
    case VAL_DOUBLE: printf("double %g\n", v->d); break;
    case VAL_STR:    printf("string \"%s\"\n", v->s); break;
    }
}

int main(void) {
    Value vals[] = {
        {.kind = VAL_INT, .i = 42},
        {.kind = VAL_DOUBLE, .d = 2.5},
        {.kind = VAL_STR, .s = "hi"},
    };
    for (int i = 0; i < 3; i++) print_value(&vals[i]);
    printf("sizeof(Value) = %zu\n", sizeof(Value));   // 16: tag + padding + 8-byte union
    return 0;
}
```

Nothing enforces the tag: reading `v->d` when `kind` is `VAL_INT` compiles fine. Keep the tag and the data in sync by only building values through small constructor functions, and always `switch` on the tag. This pattern is everywhere: interpreter values, JSON nodes, events, AST nodes, network messages.

> **Performance:** a tagged union is the right shape for a *description* (a node in a procedural texture graph, a command in a render queue, a parsed value). It is the wrong shape for a million items processed in a tight loop, because the `switch` per element defeats branch prediction and vectorization, and every element is as big as the largest variant. Evaluate the tag once and run a specialized loop per kind (one loop for all spheres, another for all boxes), or keep each kind in its own array.

## Flexible array members

The last field of a struct may be an array with no size. The struct then acts as a header, and you allocate it with as much extra room as you need, in a single block:

```c
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    size_t len;
    int data[];      // flexible array member: must be last, contributes 0 to sizeof
} IntArray;

int main(void) {
    size_t n = 5;
    IntArray *a = malloc(sizeof *a + n * sizeof a->data[0]);   // header + n ints
    if (!a) return 1;
    a->len = n;
    for (size_t i = 0; i < n; i++) a->data[i] = (int)(i * 10);
    printf("sizeof header %zu, last %d\n", sizeof(IntArray), a->data[a->len - 1]);
    free(a);         // one block, one free
    return 0;
}
```

`malloc(bytes)` asks the heap for memory and `free` gives it back; the next chapter covers both in depth. The benefit here is one allocation instead of two (header plus separate array), with the data right next to its length. You cannot declare such a struct as a normal variable or put it in an array; it only makes sense through a pointer.

## Methods without classes

C has no methods, but the pattern is easy to fake and is how every large C codebase is organized: a struct, plus functions named after it that take a pointer to it as the first argument. That pointer is `this`, made explicit.

```c
#include <stdio.h>

typedef struct {
    unsigned state;     // xorshift32 state: must never be 0
    unsigned draws;
} Rng;

void rng_init(Rng *r, unsigned seed) {         // the "constructor"
    r->state = seed ? seed : 0x9E3779B9u;
    r->draws = 0;
}

unsigned rng_next(Rng *r) {                    // a mutating "method"
    unsigned x = r->state;
    x ^= x << 13;
    x ^= x >> 17;
    x ^= x << 5;
    r->draws++;
    return r->state = x;
}

float rng_float01(Rng *r) {                    // [0, 1): the top 24 bits / 2^24
    return (rng_next(r) >> 8) * (1.0f / 16777216.0f);
}

unsigned rng_draws(const Rng *r) {             // const "this": a read-only method
    return r->draws;
}

int main(void) {
    Rng terrain, foliage;                      // independent streams, same seed: same world
    rng_init(&terrain, 42);
    rng_init(&foliage, 42);
    float a = rng_float01(&terrain), b = rng_float01(&foliage);
    printf("%d %u\n", a == b, rng_draws(&terrain));   // 1 1: deterministic per seed
    return 0;
}
```

The `typename_verb` prefix stands in for both the class and the namespace. Taking `const Rng *` is the equivalent of a method that does not mutate `this`. An RNG is also the best argument for explicit state: with the generator in a struct you pass around, every procedural generator draws from the stream you give it, and the same seed rebuilds the same world, which a hidden global `rand()` cannot promise. Encapsulation, when you want it, comes from declaring only `typedef struct rng Rng;` in a header and keeping the field list in the `.c` file, so that other files can only hold pointers and call your functions (an *opaque type*, covered with multi-file programs).

The starter defines `typedef struct { double x, y; } Point;`. Write:

- `double point_dist(Point a, Point b)`: the Euclidean distance (`sqrt` from `<math.h>`).
- `Point point_mid(Point a, Point b)`: the midpoint.
- `Point point_add(Point a, Point b)`: the component-wise sum.

Structs are small here, so pass and return them by value.
--- starter
#include <math.h>

typedef struct {
    double x, y;
} Point;

double point_dist(Point a, Point b) {
    return 0;
}

Point point_mid(Point a, Point b) {
    return a;
}

Point point_add(Point a, Point b) {
    return a;
}
--- solution
#include <math.h>

typedef struct {
    double x, y;
} Point;

double point_dist(Point a, Point b) {
    double dx = a.x - b.x, dy = a.y - b.y;
    return sqrt(dx * dx + dy * dy);
}

Point point_mid(Point a, Point b) {
    return (Point){(a.x + b.x) / 2, (a.y + b.y) / 2};
}

Point point_add(Point a, Point b) {
    return (Point){.x = a.x + b.x, .y = a.y + b.y};
}
--- hint
A compound literal builds the return value in one line: `return (Point){.x = ..., .y = ...};`
--- harness
int main(void) {
    Point o = {0, 0}, p = {3, 4}, q = {-1, 2.5};
    CHECK_DBL(point_dist(o, p), 5);
    CHECK_DBL(point_dist(p, p), 0);
    CHECK_DBL(point_dist(q, p), sqrt(16 + 2.25));
    Point m = point_mid(o, p);
    CHECK_DBL(m.x, 1.5);
    CHECK_DBL(m.y, 2);
    m = point_mid(q, q);
    CHECK_DBL(m.x, -1);
    CHECK_DBL(m.y, 2.5);
    Point s = point_add(p, q);
    CHECK_DBL(s.x, 2);
    CHECK_DBL(s.y, 6.5);
    CHECK_DBL(p.x, 3);   // the inputs were copies, untouched
    return TEST_END();
}

=== exercise rects
title: Rectangles by pointer
--- prompt
The starter defines `typedef struct { int x, y, w, h; } Rect;`: the top-left corner and a size. A rectangle covers the half-open ranges `x <= px < x + w` and `y <= py < y + h`, so a rect of width 0 contains nothing.

Write:

- `int rect_contains(const Rect *r, int px, int py)`: 1 if the point is inside, else 0.
- `int rect_intersect(const Rect *a, const Rect *b, Rect *out)`: if the two overlap with a positive area, store the overlapping rectangle in `*out` and return 1. Otherwise return 0 and do not touch `*out`. Rectangles that only share an edge do not overlap.
--- starter
typedef struct {
    int x, y, w, h;
} Rect;

int rect_contains(const Rect *r, int px, int py) {
    return 0;
}

int rect_intersect(const Rect *a, const Rect *b, Rect *out) {
    return 0;
}
--- solution
typedef struct {
    int x, y, w, h;
} Rect;

int rect_contains(const Rect *r, int px, int py) {
    return px >= r->x && px < r->x + r->w && py >= r->y && py < r->y + r->h;
}

static int max_i(int a, int b) { return a > b ? a : b; }
static int min_i(int a, int b) { return a < b ? a : b; }

int rect_intersect(const Rect *a, const Rect *b, Rect *out) {
    int x0 = max_i(a->x, b->x), y0 = max_i(a->y, b->y);
    int x1 = min_i(a->x + a->w, b->x + b->w), y1 = min_i(a->y + a->h, b->y + b->h);
    if (x1 <= x0 || y1 <= y0) return 0;
    *out = (Rect){x0, y0, x1 - x0, y1 - y0};
    return 1;
}
--- hint
Work with edges: the overlap's left edge is the larger of the two left edges, its right edge the smaller of the two right edges (`x + w`). Same for top and bottom.
--- hint
If right <= left or bottom <= top, there is no overlap.
--- harness
int main(void) {
    Rect r = {10, 20, 5, 3};
    CHECK_INT(rect_contains(&r, 10, 20), 1);
    CHECK_INT(rect_contains(&r, 14, 22), 1);
    CHECK_INT(rect_contains(&r, 15, 22), 0);
    CHECK_INT(rect_contains(&r, 14, 23), 0);
    CHECK_INT(rect_contains(&r, 9, 21), 0);
    CHECK_INT(rect_contains(&r, 12, 19), 0);
    Rect empty = {0, 0, 0, 10};
    CHECK_INT(rect_contains(&empty, 0, 0), 0);

    Rect a = {0, 0, 10, 10}, b = {5, 5, 10, 10}, out = {-1, -1, -1, -1};
    CHECK_INT(rect_intersect(&a, &b, &out), 1);
    CHECK(out.x == 5 && out.y == 5 && out.w == 5 && out.h == 5);

    Rect inner = {2, 3, 4, 1};
    CHECK_INT(rect_intersect(&a, &inner, &out), 1);
    CHECK(out.x == 2 && out.y == 3 && out.w == 4 && out.h == 1);
    CHECK_INT(rect_intersect(&inner, &a, &out), 1);
    CHECK(out.x == 2 && out.y == 3 && out.w == 4 && out.h == 1);

    Rect edge = {10, 0, 5, 5}, keep = {7, 7, 7, 7};
    out = keep;
    CHECK_INT(rect_intersect(&a, &edge, &out), 0);
    CHECK(out.x == 7 && out.w == 7);
    Rect far = {-20, -20, 5, 5};
    CHECK_INT(rect_intersect(&a, &far, &out), 0);
    CHECK_INT(rect_intersect(&a, &empty, &out), 0);
    Rect neg = {-5, -5, 7, 20};
    CHECK_INT(rect_intersect(&neg, &a, &out), 1);
    CHECK(out.x == 0 && out.y == 0 && out.w == 2 && out.h == 10);
    return TEST_END();
}

title: A stack with methods

--- solution
#define STACK_CAP 8
typedef struct {
    int items[STACK_CAP];
    int count;
} Stack;

void stack_init(Stack *s) {
    s->count = 0;
}

int stack_push(Stack *s, int v) {
    if (s->count == STACK_CAP) return 0;
    s->items[s->count++] = v;
    return 1;
}

int stack_pop(Stack *s, int *out) {
    if (s->count == 0) return 0;
    *out = s->items[--s->count];
    return 1;
}

int stack_peek(const Stack *s, int *out) {
    if (s->count == 0) return 0;
    *out = s->items[s->count - 1];
    return 1;
}

int stack_size(const Stack *s) {

typedef enum { SHAPE_CIRCLE, SHAPE_RECT, SHAPE_TRIANGLE } ShapeKind;

typedef struct {
    ShapeKind kind;
    union {
        struct { double r; } circle;
        struct { double w, h; } rect;
        struct { double a, b, c; } tri;
    };
} Shape;

Shape make_circle(double r) {
    return (Shape){.kind = SHAPE_CIRCLE, .circle = {r}};
}

Shape make_rect(double w, double h) {
    return (Shape){.kind = SHAPE_RECT, .rect = {w, h}};
}

Shape make_tri(double a, double b, double c) {
    return (Shape){.kind = SHAPE_TRIANGLE, .tri = {a, b, c}};
}

double shape_area(const Shape *s) {
    switch (s->kind) {
    case SHAPE_CIRCLE:
        return PI * s->circle.r * s->circle.r;
    case SHAPE_RECT:
        return s->rect.w * s->rect.h;
    case SHAPE_TRIANGLE: {
        double a = s->tri.a, b = s->tri.b, c = s->tri.c;
        double p = (a + b + c) / 2;
        return sqrt(p * (p - a) * (p - b) * (p - c));
    }
    }
    return 0;
}

double shape_perimeter(const Shape *s) {
    switch (s->kind) {
    case SHAPE_CIRCLE: return 2 * PI * s->circle.r;
    case SHAPE_RECT: return 2 * (s->rect.w + s->rect.h);
    case SHAPE_TRIANGLE: return s->tri.a + s->tri.b + s->tri.c;
    }
    return 0;
}

#include <stdio.h>
#include <string.h>

typedef struct {
    char name[16];
    int score;
} Player;

static int before(const Player *a, const Player *b) {
    if (a->score != b->score) return a->score > b->score;
    return strcmp(a->name, b->name) < 0;
}

int main(void) {
    Player p[100];
    int n;
    if (scanf("%d", &n) != 1 || n < 1 || n > 100) return 1;
    for (int i = 0; i < n; i++)
        if (scanf("%15s %d", p[i].name, &p[i].score) != 2) return 1;

    for (int i = 1; i < n; i++) {
        Player key = p[i];
        int j = i;
        while (j > 0 && before(&key, &p[j - 1])) {
            p[j] = p[j - 1];
            j--;
        }
        p[j] = key;
    }
    for (int i = 0; i < n; i++)
        printf("%d. %s %d\n", i + 1, p[i].name, p[i].score);
    return 0;
}

#include <stdlib.h>
#include <string.h>

typedef struct {
    size_t len;
    char chars[];
} Str;

static Str *str_alloc(size_t len) {
    Str *s = malloc(sizeof *s + len + 1);
    if (s) s->len = len;
    return s;
}

Str *str_new(const char *s) {
    size_t len = strlen(s);
    Str *r = str_alloc(len);
    if (!r) return NULL;
    memcpy(r->chars, s, len + 1);
    return r;
}

Str *str_concat(const Str *a, const Str *b) {
    Str *r = str_alloc(a->len + b->len);
    if (!r) return NULL;
    memcpy(r->chars, a->chars, a->len);
    memcpy(r->chars + a->len, b->chars, b->len + 1);
    return r;
}

int str_equal(const Str *a, const Str *b) {
    return a->len == b->len && memcmp(a->chars, b->chars, a->len) == 0;
}
--- hint
The size to allocate is `sizeof(Str) + len + 1`: the header (just the `size_t` here), the characters, and the terminator.

=== exercise normals
title: Smooth vertex normals
continues: vec3
--- prompt
A mesh is an array of vertices plus an **index buffer**: every three indices name the vertices of one triangle, so a vertex shared by six triangles is stored once. Lighting needs a normal per vertex. The standard *smooth normal* is: for every triangle, add its (unnormalized) cross product to the normal of each of its three vertices, then normalize every vertex normal. Using the unnormalized cross product weights each face by its area, which is what you want.

The stubs add

```c
typedef struct { Vec3 pos; Vec3 normal; } Vertex;
void compute_normals(Vertex *verts, int nv, const unsigned *idx, int ni);
```

`ni` is a multiple of 3. Overwrite every `normal` (zero them first: they may hold garbage), and leave vertices that are in no triangle at `(0, 0, 0)`.

The shape of this loop matters: one pass over the index buffer, accumulating, then one pass over the vertices, normalizing. Searching for "all triangles that touch vertex v" for each vertex would be O(vertices x triangles); the tests use a 300x300 height-field grid with 178,802 triangles.
--- stubs

}
--- solution
#include <math.h>

typedef struct { float x, y, z; } Vec3;

Vec3 v3(float x, float y, float z) { Vec3 r = {x, y, z}; return r; }
Vec3 v3_add(Vec3 a, Vec3 b) { return v3(a.x + b.x, a.y + b.y, a.z + b.z); }
Vec3 v3_sub(Vec3 a, Vec3 b) { return v3(a.x - b.x, a.y - b.y, a.z - b.z); }
Vec3 v3_scale(Vec3 a, float k) { return v3(a.x * k, a.y * k, a.z * k); }
float v3_dot(Vec3 a, Vec3 b) { return a.x * b.x + a.y * b.y + a.z * b.z; }

Vec3 v3_cross(Vec3 a, Vec3 b) {
    return v3(a.y * b.z - a.z * b.y, a.z * b.x - a.x * b.z, a.x * b.y - a.y * b.x);
}

Vec3 v3_normalize(Vec3 a) {
    float len = sqrtf(v3_dot(a, a));
    return len > 0 ? v3_scale(a, 1.0f / len) : v3(0, 0, 0);
}

Vec3 tri_normal(Vec3 a, Vec3 b, Vec3 c) {
    return v3_normalize(v3_cross(v3_sub(b, a), v3_sub(c, a)));
}

typedef struct { Vec3 pos; Vec3 normal; } Vertex;

void compute_normals(Vertex *verts, int nv, const unsigned *idx, int ni) {
    for (int i = 0; i < nv; i++) verts[i].normal = v3(0, 0, 0);
    for (int t = 0; t + 2 < ni; t += 3) {
        Vertex *a = &verts[idx[t]], *b = &verts[idx[t + 1]], *c = &verts[idx[t + 2]];
        Vec3 n = v3_cross(v3_sub(b->pos, a->pos), v3_sub(c->pos, a->pos));
        a->normal = v3_add(a->normal, n);
        b->normal = v3_add(b->normal, n);
        c->normal = v3_add(c->normal, n);
    }
    for (int i = 0; i < nv; i++) verts[i].normal = v3_normalize(verts[i].normal);
}
--- hint
Three loops: zero all normals; for each triangle `t` (step 3), compute the cross product once and add it to all three vertices; normalize all normals.

=== exercise soabounds
title: Bounds over struct-of-arrays
continues: normals
--- prompt
Your `Vertex` is an *array of structs* (AoS): position and normal side by side, 24 bytes per vertex, often more once UVs and colors join. A pass that only needs positions still drags every normal through the cache. The alternative layout is a *struct of arrays* (SoA): one array per component.

Add:

- `void split_positions(const Vertex *v, int n, float *xs, float *ys, float *zs)`: copy the positions out into three arrays.
- `typedef struct { Vec3 min, max; } Box3;` (in the stubs) and `Box3 bounds_soa(const float *xs, const float *ys, const float *zs, int n)`: the axis-aligned bounding box of `n >= 1` points.

> **Performance:** in the SoA loop each of the three streams is read in order, every byte fetched is used, and the compiler can process 8 floats per instruction with AVX. The AoS loop over the same data reads 24 bytes to use 12. AoS is still the right layout for data that is always used together (a GPU vertex buffer is usually interleaved for that reason); SoA wins for hot loops that touch one or two fields of many elements: bounds, culling, particle physics.
--- stubs

}

Box3 bounds_soa(const float *xs, const float *ys, const float *zs, int n) {
    Box3 b = {{0, 0, 0}, {0, 0, 0}};
    return b;
}
--- solution
#include <math.h>

typedef struct { float x, y, z; } Vec3;

Vec3 v3(float x, float y, float z) { Vec3 r = {x, y, z}; return r; }
Vec3 v3_add(Vec3 a, Vec3 b) { return v3(a.x + b.x, a.y + b.y, a.z + b.z); }
Vec3 v3_sub(Vec3 a, Vec3 b) { return v3(a.x - b.x, a.y - b.y, a.z - b.z); }
Vec3 v3_scale(Vec3 a, float k) { return v3(a.x * k, a.y * k, a.z * k); }
float v3_dot(Vec3 a, Vec3 b) { return a.x * b.x + a.y * b.y + a.z * b.z; }

Vec3 v3_cross(Vec3 a, Vec3 b) {
    return v3(a.y * b.z - a.z * b.y, a.z * b.x - a.x * b.z, a.x * b.y - a.y * b.x);
}

Vec3 v3_normalize(Vec3 a) {
    float len = sqrtf(v3_dot(a, a));
    return len > 0 ? v3_scale(a, 1.0f / len) : v3(0, 0, 0);
}

Vec3 tri_normal(Vec3 a, Vec3 b, Vec3 c) {
    return v3_normalize(v3_cross(v3_sub(b, a), v3_sub(c, a)));
}

typedef struct { Vec3 pos; Vec3 normal; } Vertex;

void compute_normals(Vertex *verts, int nv, const unsigned *idx, int ni) {
    for (int i = 0; i < nv; i++) verts[i].normal = v3(0, 0, 0);
    for (int t = 0; t + 2 < ni; t += 3) {
        Vertex *a = &verts[idx[t]], *b = &verts[idx[t + 1]], *c = &verts[idx[t + 2]];
        Vec3 n = v3_cross(v3_sub(b->pos, a->pos), v3_sub(c->pos, a->pos));
        a->normal = v3_add(a->normal, n);
        b->normal = v3_add(b->normal, n);
        c->normal = v3_add(c->normal, n);
    }
    for (int i = 0; i < nv; i++) verts[i].normal = v3_normalize(verts[i].normal);
}

typedef struct { Vec3 min, max; } Box3;

void split_positions(const Vertex *v, int n, float *xs, float *ys, float *zs) {
    for (int i = 0; i < n; i++) {
        xs[i] = v[i].pos.x;
        ys[i] = v[i].pos.y;
        zs[i] = v[i].pos.z;
    }
}

static void range(const float *a, int n, float *lo, float *hi) {
    float l = a[0], h = a[0];
    for (int i = 1; i < n; i++) {
        l = a[i] < l ? a[i] : l;
        h = a[i] > h ? a[i] : h;
    }
    *lo = l;
    *hi = h;
}

Box3 bounds_soa(const float *xs, const float *ys, const float *zs, int n) {
    Box3 b;
    range(xs, n, &b.min.x, &b.max.x);       // one stream at a time
    range(ys, n, &b.min.y, &b.max.y);
    range(zs, n, &b.min.z, &b.max.z);
    return b;
}
--- hint
Start min and max at the first element, not at 0 (a mesh entirely at negative x would otherwise get `max.x = 0`).
--- hint
A helper that finds the range of one array, called three times, is both simple and the fastest shape: each call is one sequential stream.
--- harness
static Vertex vx(float x, float y, float z) { Vertex v = {{x, y, z}, {0, 0, 0}}; return v; }
int main(void) {
    Vertex v[4] = {vx(1, -2, 3), vx(-4, 5, 6), vx(7, 8, -9), vx(0, 0, 0)};
    float xs[4], ys[4], zs[4];
    split_positions(v, 4, xs, ys, zs);
    CHECK_DBL(xs[1], -4); CHECK_DBL(ys[2], 8); CHECK_DBL(zs[2], -9); CHECK_DBL(zs[0], 3);
    Box3 b = bounds_soa(xs, ys, zs, 4);
    CHECK_DBL(b.min.x, -4); CHECK_DBL(b.min.y, -2); CHECK_DBL(b.min.z, -9);
    CHECK_DBL(b.max.x, 7); CHECK_DBL(b.max.y, 8); CHECK_DBL(b.max.z, 6);
    float nx[2] = {-5, -3}, ny[2] = {-1, -2}, nz[2] = {-7, -7};
    b = bounds_soa(nx, ny, nz, 2);
    CHECK_DBL(b.max.x, -3); CHECK_DBL(b.max.y, -1); CHECK_DBL(b.max.z, -7); CHECK_DBL(b.min.x, -5);
    b = bounds_soa(xs + 2, ys + 2, zs + 2, 1);
    CHECK_DBL(b.min.x, 7); CHECK_DBL(b.max.x, 7); CHECK_DBL(b.min.z, -9);
    int n = 1000000;
    Vertex *big = malloc((size_t)n * sizeof *big);
    float *X = malloc(n * sizeof *X), *Y = malloc(n * sizeof *Y), *Z = malloc(n * sizeof *Z);
    for (int i = 0; i < n; i++) big[i] = vx(sinf((float)i) * 10, (float)(i % 1000) - 500, (float)i * 0.001f);
    split_positions(big, n, X, Y, Z);

    CHECK(b.max.x > 9.99f && b.max.x <= 10.0f && b.min.x < -9.99f);
    free(big); free(X); free(Y); free(Z);
    return TEST_END();

=== exercise texnode
title: Procedural textures as a tagged union
--- prompt
A procedural texture is a small description, not a stored image. Describe a few kinds with a tagged union (the starter has the types):

```c
typedef enum { TEX_SOLID, TEX_CHECKER, TEX_STRIPES } TexKind;
typedef struct {
    TexKind kind;
    uint32_t a, b;         // colors; SOLID uses only a
    int size;              // CHECKER: square size, STRIPES: stripe height (both >= 1)
} Tex;
```

(No union is needed when every kind shares the same fields; real texture nodes have different parameters per kind, which is when the union in the body text pays off.) Write:

- `uint32_t tex_sample(const Tex *t, int x, int y)`: SOLID is `a`. CHECKER is `a` when `floor(x / size) + floor(y / size)` is even, else `b`. STRIPES is `a` when `floor(y / size)` is even, else `b`.
- `void tex_fill(const Tex *t, uint32_t *img, int w, int h, int ox, int oy)`: fill a row-major `w x h` image with the texture sampled at `(ox + x, oy + y)`.

The trap is **negative coordinates**: textures tile across the origin, and C's `/` truncates toward zero, so `-1 / 4` is 0, the same square as `1 / 4`, and the checker pattern gets a double-wide row at the axis. Write a `floor_div` helper.

--- solution
#include <stdio.h>
#include <stdint.h>

typedef enum { TEX_SOLID, TEX_CHECKER, TEX_STRIPES } TexKind;
typedef struct {
    TexKind kind;
    uint32_t a, b;
    int size;
} Tex;

static int floor_div(int a, int b) {           // b > 0
    int q = a / b;
    return (a % b != 0 && a < 0) ? q - 1 : q;
}

uint32_t tex_sample(const Tex *t, int x, int y) {
    switch (t->kind) {
    case TEX_CHECKER:
        return ((floor_div(x, t->size) + floor_div(y, t->size)) & 1) ? t->b : t->a;
    case TEX_STRIPES:
        return (floor_div(y, t->size) & 1) ? t->b : t->a;
    case TEX_SOLID:
    default:
        return t->a;
    }
}

void tex_fill(const Tex *t, uint32_t *img, int w, int h, int ox, int oy) {
    switch (t->kind) {
    case TEX_CHECKER:
        for (int y = 0; y < h; y++) {
            int cy = floor_div(oy + y, t->size);
            for (int x = 0; x < w; x++)
                img[y * w + x] = ((floor_div(ox + x, t->size) + cy) & 1) ? t->b : t->a;
        }
        break;
    case TEX_STRIPES:
        for (int y = 0; y < h; y++) {
            uint32_t c = (floor_div(oy + y, t->size) & 1) ? t->b : t->a;   // one color per row
            for (int x = 0; x < w; x++) img[y * w + x] = c;
        }
        break;
    case TEX_SOLID:
    default:
        for (int i = 0; i < w * h; i++) img[i] = t->a;
        break;
    }
}

int main(void) {
    Tex t = {TEX_CHECKER, 0xFF2020E0u, 0xFFF0F0F0u, 4};
    static uint32_t img[32 * 32];
    tex_fill(&t, img, 32, 32, -16, -16);
    printf("P3\n32 32\n255\n");
    for (int i = 0; i < 32 * 32; i++)
        printf("%u %u %u\n", img[i] & 255, img[i] >> 8 & 255, img[i] >> 16 & 255);
    return 0;
}
--- hint
`floor_div(a, b)` for `b > 0`: take `q = a / b`; if `a` is negative and not an exact multiple, subtract 1.
--- hint
`& 1` tests odd/even and, unlike `% 2`, gives 1 for odd negative numbers too (`-3 % 2` is `-1`).
--- harness
int main(void) {
    Tex s = {TEX_SOLID, 7, 9, 1};
    CHECK_UINT(tex_sample(&s, -5, 100), 7);
    Tex c = {TEX_CHECKER, 1, 2, 4};
    CHECK_UINT(tex_sample(&c, 0, 0), 1);
    CHECK_UINT(tex_sample(&c, 3, 3), 1);
    CHECK_UINT(tex_sample(&c, 4, 0), 2);
    CHECK_UINT(tex_sample(&c, 4, 4), 1);
    CHECK_UINT(tex_sample(&c, -1, 0), 2);
    CHECK_UINT(tex_sample(&c, -4, 0), 2);
    CHECK_UINT(tex_sample(&c, -5, 0), 1);
    CHECK_UINT(tex_sample(&c, -1, -1), 1);
    Tex st = {TEX_STRIPES, 10, 20, 3};
    CHECK_UINT(tex_sample(&st, 99, 2), 10);
    CHECK_UINT(tex_sample(&st, 99, 3), 20);
    CHECK_UINT(tex_sample(&st, 99, -1), 20);
    CHECK_UINT(tex_sample(&st, 99, -3), 20);
    CHECK_UINT(tex_sample(&st, 99, -4), 10);
    int W = 37, H = 23, bad = 0;
    uint32_t img[37 * 23];
    Tex all[3] = {s, c, st};
    for (int k = 0; k < 3; k++)
        for (int o = -40; o <= 40; o += 20) {

            if (img[y * 16 + x] != img[(y / 4 * 4) * 16 + x / 4 * 4]) seams++;
    CHECK_INT(seams, 0);
    CHECK(img[0] != img[4]);
    return TEST_END();
}
